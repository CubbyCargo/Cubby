function parseBlastSheet(rows){ const blast = {avgOpenRate:null, newInbound:null, campaigns:[]}; if(!rows || rows.length < 2) return blast; // Find header row let headerRowIdx = -1; for(let i = 0; i < rows.length; i++){ const row = rows[i].map(c => String(c || '').trim().toLowerCase()); if(row.some(c => c.includes('campaign'))){ headerRowIdx = i; break; } } if(headerRowIdx < 0) return blast; const headers = rows[headerRowIdx].map(c => String(c || '').trim().toLowerCase() ); const ci = { camp: headers.findIndex(h => h.includes('campaign')), rec: headers.findIndex(h => h.includes('recipient')), open: headers.findIndex(h => h.includes('open rate')), inter: headers.findIndex(h => h.includes('interact')), inbound: headers.findIndex(h => h.includes('inbound')) }; let inboundSum = 0; let totalRowFound = false; for(let i = headerRowIdx + 1; i < rows.length; i++){ const row = rows[i]; const name = String(row[ci.camp] || '').trim(); // Detect TOTAL row if(name.toLowerCase() === 'total'){ totalRowFound = true; if(ci.open >= 0){ const v = String(row[ci.open] || '').replace(/[%\s]/g, ''); if(v) blast.avgOpenRate = parseFloat(v); } if(ci.inbound >= 0){ const v = String(row[ci.inbound] || '').replace(/[^0-9.-]/g, ''); if(v) blast.newInbound = parseInt(v, 10); } // Don't add TOTAL to campaign table continue; } if(!name) continue; // Add campaign inbound values as fallback if(ci.inbound >= 0){ const v = String(row[ci.inbound] || '').replace(/[^0-9.-]/g, ''); if(v) inboundSum += parseInt(v, 10) || 0; } blast.campaigns.push({ name, recipients: parseInt(String(row[ci.rec] || '0').replace(/[^0-9]/g, '')) || 0, openRate: parseFloat(String(row[ci.open] || '0').replace(/[%\s]/g, '')) || 0, interactionRate: parseFloat(String(row[ci.inter] || '0').replace(/[%\s]/g, '')) || 0 }); } // If there was no usable TOTAL inbound, calculate it if(blast.newInbound === null){ blast.newInbound = inboundSum; } // If there was no TOTAL open rate, calculate average campaign open rate if(blast.avgOpenRate === null && blast.campaigns.length){ blast.avgOpenRate = Math.round( blast.campaigns.reduce((sum, c) => sum + c.openRate, 0) / blast.campaigns.length ); } return blast; }
